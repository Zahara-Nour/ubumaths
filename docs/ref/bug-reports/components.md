## BugReportFAB

Floating Action Button pour ouvrir le formulaire de bug report.

### Fichier

`src/lib/components/bug-reports/BugReportFAB.svelte`

### Props

| Prop      | Type         | Default  | Description     |
| --------- | ------------ | -------- | --------------- |
| `onclick` | `() => void` | required | Handler du clic |

### Usage

```svelte
<script lang="ts">
	import BugReportFAB from '$lib/components/bug-reports/BugReportFAB.svelte';

	let dialogOpen = $state(false);
</script>

<BugReportFAB onclick={() => (dialogOpen = true)} />
```

### Comportement

- Position fixe en bas a droite (`fixed bottom-4 right-4`)
- Z-index eleve pour etre au-dessus du contenu
- Icone bug (:bug:) avec effet hover
- Tooltip "Signaler un probleme"

### Styles

```css
/* Position et apparence */
fixed bottom-4 right-4 z-50
rounded-full p-3
bg-primary text-primary-foreground
shadow-lg hover:shadow-xl
transition-all duration-200
```

## BugReportDialog

Formulaire complet de soumission de bug report.

### Fichier

`src/lib/components/bug-reports/BugReportDialog.svelte`

### Props

| Prop                 | Type                      | Default  | Description                     |
| -------------------- | ------------------------- | -------- | ------------------------------- |
| `open`               | `boolean`                 | required | Controle l'ouverture (bindable) |
| `onOpenChange`       | `(open: boolean) => void` | required | Callback de changement          |
| `autoGenerated`      | `boolean`                 | `false`  | True si rapport automatique     |
| `initialDescription` | `string`                  | `''`     | Description pre-remplie         |

### State interne

```typescript
let category = $state<BugReportCategory>('bug');
let severity = $state<BugReportSeverity>('medium');
let title = $state('');
let description = $state(initialDescription);
let screenshotFile = $state<File | null>(null);
let screenshotPreview = $state<string | null>(null);
let isSubmitting = $state(false);
```

### Validation derivee

```typescript
const titleValid = $derived(title.trim().length >= 5 && title.trim().length <= 200);
const descriptionValid = $derived(
	description.trim().length >= 20 && description.trim().length <= 5000
);
const isValid = $derived(titleValid && descriptionValid);
```

### Usage

```svelte
<script lang="ts">
	import BugReportDialog from '$lib/components/bug-reports/BugReportDialog.svelte';

	let dialogOpen = $state(false);
</script>

<BugReportDialog bind:open={dialogOpen} onOpenChange={(open) => (dialogOpen = open)} />
```

### Usage avec pre-remplissage (freeze)

```svelte
<BugReportDialog
	bind:open={dialogOpen}
	onOpenChange={(open) => (dialogOpen = open)}
	autoGenerated={true}
	initialDescription="L'application s'est figee pendant environ 15 secondes..."
/>
```

### Champs du formulaire

| Champ       | Type     | Component             | Validation        |
| ----------- | -------- | --------------------- | ----------------- |
| Category    | Select   | `MySelect`            | Required          |
| Severity    | Radio    | `RadioGroup`          | Default: medium   |
| Title       | Text     | `Input`               | 5-200 chars       |
| Description | Textarea | `Textarea`            | 20-5000 chars     |
| Screenshot  | File     | `<input type="file">` | Optional, 5MB max |

### Contexte capture automatiquement

```typescript
const pageUrl = $derived(typeof window !== 'undefined' ? window.location.href : '');
const userAgent = $derived(typeof navigator !== 'undefined' ? navigator.userAgent : '');
const viewportSize = $derived(
	typeof window !== 'undefined' ? `${window.innerWidth}x${window.innerHeight}` : ''
);
```

### Flux de soumission

```typescript
async function handleSubmit() {
	// 1. Get freeze detection context
	const freezeContext = getFreezeDetectionContext();

	// 2. Create bug report
	const response = await fetch('/api/bug-reports', {
		method: 'POST',
		body: JSON.stringify({
			category,
			severity,
			title,
			description,
			pageUrl,
			userAgent,
			viewportSize,
			sessionContext: { ...freezeContext, capturedAt: new Date().toISOString() },
			autoGenerated
		})
	});

	// 3. Upload screenshot if provided
	if (screenshotFile && report.id) {
		const formData = new FormData();
		formData.append('image', screenshotFile);
		await fetch(`/api/bug-reports/${report.id}/screenshot`, {
			method: 'POST',
			body: formData
		});
	}

	// 4. Show success toast and close
	toaster.success('Rapport envoye avec succes');
	onOpenChange(false);
}
```

## BugReportCard

Carte affichant un resume de rapport avec support de selection.

### Fichier

`src/lib/components/bug-reports/BugReportCard.svelte`

### Props

| Prop                | Type                        | Default  | Description                    |
| ------------------- | --------------------------- | -------- | ------------------------------ |
| `report`            | `BugReportWithAuthor`       | required | Donnees du rapport             |
| `onclick`           | `() => void`                | -        | Handler de clic (optionnel)    |
| `showAuthor`        | `boolean`                   | `false`  | Afficher l'auteur (admin view) |
| `selectable`        | `boolean`                   | `false`  | Affiche une checkbox           |
| `selected`          | `boolean`                   | `false`  | Etat de selection              |
| `onSelectionChange` | `(selected: boolean)=>void` | -        | Callback de changement         |

### Usage

```svelte
<script lang="ts">
	import BugReportCard from '$lib/components/bug-reports/BugReportCard.svelte';

	let reports: BugReportWithAuthor[] = [];
	let selectedReport = $state<BugReportWithAuthor | null>(null);
</script>

{#each reports as report (report.id)}
	<BugReportCard {report} onclick={() => (selectedReport = report)} showAuthor={isAdmin} />
{/each}
```

### Usage avec selection (admin batch operations)

```svelte
<script lang="ts">
	let selectedIds = $state<Set<string>>(new Set());
</script>

{#each reports as report (report.id)}
	<BugReportCard
		{report}
		selectable={true}
		selected={selectedIds.has(report.id)}
		onSelectionChange={(selected) => {
			if (selected) selectedIds.add(report.id);
			else selectedIds.delete(report.id);
			selectedIds = selectedIds; // trigger reactivity
		}}
	/>
{/each}
```

## BugReportList

Liste filtrable de rapports avec support de selection en lot.

### Fichier

`src/lib/components/bug-reports/BugReportList.svelte`

### Props

| Prop                | Type                                    | Default     | Description                |
| ------------------- | --------------------------------------- | ----------- | -------------------------- |
| `reports`           | `BugReportWithAuthor[]`                 | required    | Liste des rapports         |
| `loading`           | `boolean`                               | `false`     | Etat de chargement         |
| `showFilters`       | `boolean`                               | `true`      | Afficher les filtres       |
| `onReportClick`     | `(report: BugReportWithAuthor) => void` | -           | Callback clic rapport      |
| `onFilterChange`    | `(filters) => void`                     | -           | Callback changement filtre |
| `selectable`        | `boolean`                               | `false`     | Active les checkboxes      |
| `selectedIds`       | `Set<string>`                           | `new Set()` | IDs selectionnes           |
| `onSelectionChange` | `(ids: Set<string>) => void`            | -           | Callback selection         |

### Filtres

```typescript
let statusFilter = $state<BugReportStatus | 'all'>('all');
let categoryFilter = $state<BugReportCategory | 'all'>('all');
let severityFilter = $state<BugReportSeverity | 'all'>('all'); // Admin only

const filteredReports = $derived(
	reports.filter((r) => {
		if (statusFilter !== 'all' && r.status !== statusFilter) return false;
		if (categoryFilter !== 'all' && r.category !== categoryFilter) return false;
		if (severityFilter !== 'all' && r.severity !== severityFilter) return false;
		return true;
	})
);
```

### Usage

```svelte
<script lang="ts">
	import BugReportList from '$lib/components/bug-reports/BugReportList.svelte';

	let { data } = $props();
	let selectedReport = $state<BugReportWithAuthor | null>(null);
</script>

<BugReportList reports={data.reports} onReportClick={(report) => (selectedReport = report)} />
```

### Usage avec selection en lot (admin)

```svelte
<script lang="ts">
	import BugReportList from '$lib/components/bug-reports/BugReportList.svelte';

	let { data } = $props();
	let selectedIds = $state<Set<string>>(new Set());

	function handleBatchDelete() {
		// Call batch API
		fetch('/api/bug-reports/batch', {
			method: 'DELETE',
			body: JSON.stringify({ reportIds: [...selectedIds] })
		});
	}
</script>

<BugReportList
	reports={data.reports}
	selectable={true}
	{selectedIds}
	onSelectionChange={(ids) => (selectedIds = ids)}
	onReportClick={(report) => console.log('clicked', report)}
/>

{#if selectedIds.size > 0}
	<div class="fixed bottom-4 left-1/2 -translate-x-1/2">
		<span>{selectedIds.size} selectionne(s)</span>
		<button onclick={handleBatchDelete}>Supprimer</button>
	</div>
{/if}
```

### Fonctionnalite "Tout selectionner"

Quand `selectable=true`, un header avec checkbox "Tout selectionner" apparait.

- Coche tous les rapports affiches (filtres appliques)
- Gere l'etat indeterminate (selection partielle)
- Affiche un compteur de selection

| `freezeDuration` | `number`                  | required | Duree du freeze en ms           |

### Usage (dans dashboard layout)

```svelte
<script lang="ts">
	import FreezeReportPrompt from '$lib/components/bug-reports/FreezeReportPrompt.svelte';
	import { setFreezePromptCallback } from '$lib/utils/freezeDetection';

	let freezePromptOpen = $state(false);
	let freezeDuration = $state(0);

	$effect(() => {
		setFreezePromptCallback((duration, context) => {
			freezeDuration = duration;
			freezePromptOpen = true;
		});

<FreezeReportPrompt
	bind:open={freezePromptOpen}
	onOpenChange={(open) => (freezePromptOpen = open)}

	onSubmit={() => {
		// Open full bug report dialog with pre-filled data
		bugReportDialogOpen = true;
	}}
/>
```

## ExportClaudeCodeButton

Bouton d'export avec preview pour admins.

### Fichier

`src/lib/components/bug-reports/ExportClaudeCodeButton.svelte`

### Props

| Prop       | Type      | Default  | Description          |
| ---------- | --------- | -------- | -------------------- |
| `reportId` | `string`  | required | ID du rapport        |
| `disabled` | `boolean` | `false`  | Desactiver le bouton |

### Actions disponibles

| Action   | Description                        |
| -------- | ---------------------------------- |
| Preview  | Affiche le Markdown dans un dialog |
| Copy     | Copie dans le presse-papier        |
| Download | Telecharge le fichier .md          |

### Usage

```svelte
<script lang="ts">
	import ExportClaudeCodeButton from '$lib/components/bug-reports/ExportClaudeCodeButton.svelte';
</script>

{#if profile.role === 'admin'}
	<ExportClaudeCodeButton reportId={report.id} />
{/if}
```

### Flux d'export

```typescript
async function handleExport(action: 'preview' | 'copy' | 'download') {
	const response = await fetch(`/api/bug-reports/${reportId}/export`);
	const markdown = await response.text();

	switch (action) {
		case 'preview':
			previewContent = markdown;
			previewOpen = true;
			break;
		case 'copy':
			await navigator.clipboard.writeText(markdown);
			toaster.success('Copie dans le presse-papier');
			break;
		case 'download':
			downloadFile(markdown, filename);
			break;
	}
}
```

## Integration Dashboard

### Layout principal

`src/routes/(protected)/dashboard/+layout.svelte`

```svelte
<script lang="ts">
	import BugReportFAB from '$lib/components/bug-reports/BugReportFAB.svelte';
	import BugReportDialog from '$lib/components/bug-reports/BugReportDialog.svelte';
	import FreezeReportPrompt from '$lib/components/bug-reports/FreezeReportPrompt.svelte';
	import {
		initFreezeDetection,
		cleanupFreezeDetection,
		setFreezePromptCallback,
		setAutoReportCallback
	} from '$lib/utils/freezeDetection';

	// State
	let bugReportDialogOpen = $state(false);
	let freezePromptOpen = $state(false);
	let freezeDuration = $state(0);
	let autoGeneratedDescription = $state('');

	// Initialize freeze detection
	$effect(() => {
		initFreezeDetection();

		// Callback for 15s+ freeze
		setFreezePromptCallback((duration, context) => {
			freezeDuration = duration;
			freezePromptOpen = true;
		});

		// Callback for 30s+ freeze (auto-report)
		setAutoReportCallback(async (duration, context) => {
			await fetch('/api/bug-reports', {
				method: 'POST',
				headers: { 'Content-Type': 'application/json' },
				body: JSON.stringify({
					category: 'bug',
					severity: 'high',
					title: `Application figee pendant ${Math.round(duration / 1000)}s`,
					description: 'Rapport genere automatiquement...',
					autoGenerated: true,
					sessionContext: context
				})
			});
		});

		return () => cleanupFreezeDetection();
	});
</script>

<!-- Main content -->
<slot />

<!-- Bug Report FAB -->
<BugReportFAB onclick={() => (bugReportDialogOpen = true)} />

<!-- Bug Report Dialog -->
<BugReportDialog
	bind:open={bugReportDialogOpen}
	onOpenChange={(open) => (bugReportDialogOpen = open)}
	autoGenerated={!!autoGeneratedDescription}
	initialDescription={autoGeneratedDescription}
/>

<!-- Freeze Prompt -->
<FreezeReportPrompt
	bind:open={freezePromptOpen}
	onOpenChange={(open) => (freezePromptOpen = open)}
	{freezeDuration}
	onSubmit={() => {
		autoGeneratedDescription = `L'application s'est figee pendant ${Math.round(freezeDuration / 1000)} secondes.`;
		bugReportDialogOpen = true;
	}}
/>
```
